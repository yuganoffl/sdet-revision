Lambda and predicate

Lambda and predicateJava 8 core vocabulary, in one file: FUNCTIONAL INTERFACE, LAMBDA, and PREDICATE. - FUNCTIONAL INTERFACE: an interface with exactly ONE abstract method.

LambdaAndPredicate.java

package com.sdet.studyplan.java8;

import java.util.Arrays;
import java.util.List;
import java.util.function.Predicate;
import java.util.stream.Collectors;

/**
 * Java 8 core vocabulary, in one file: FUNCTIONAL INTERFACE, LAMBDA, and
 * PREDICATE.
 *
 * - FUNCTIONAL INTERFACE: an interface with exactly ONE abstract method.
 *   Because there is only one method to implement, Java lets you supply
 *   that single method inline instead of writing a whole class.
 *   Predicate<T> is the classic example: its one method is
 *   "boolean test(T value)".
 *
 * - LAMBDA: the inline implementation of that single method.
 *   Read "number -> number % 2 == 0" as: "take a number, return whether
 *   it is even". Left of the arrow are the parameters, right is the body.
 *
 * Why a test engineer cares: Selenium's waits, stream filters, and most
 * modern Java test libraries accept lambdas, so you will read and write
 * them constantly.
 */
public class LambdaAndPredicate {

    public static void main(String[] args) {

        // A Predicate<Integer> named isEven: one input, true/false answer.
        Predicate<Integer> isEven = number -> number % 2 == 0;

        // Call its single method, test(), directly.
        System.out.println("isEven.test(4): " + isEven.test(4));
        System.out.println("isEven.test(7): " + isEven.test(7));

        // The same logic BEFORE Java 8 took an anonymous class — this is
        // what the lambda replaces (no need to write this way anymore):
        Predicate<Integer> isEvenOldStyle = new Predicate<Integer>() {
            @Override
            public boolean test(Integer number) {
                return number % 2 == 0;
            }
        };
        System.out.println("Old style:      " + isEvenOldStyle.test(4));

        // Where lambdas really shine: passing behavior into a method.
        // Stream.filter() keeps only the elements for which the predicate
        // returns true.
        List<Integer> numbers = Arrays.asList(1, 2, 3, 4, 5, 6, 7, 8);

        List<Integer> evens = numbers.stream()
                .filter(isEven) // reuse the same predicate object
                .collect(Collectors.toList());

        System.out.println("Filtered evens: " + evens);

        // Predicates combine: and(), or(), negate().
        Predicate<Integer> isPositive = number -> number > 0;
        List<Integer> mixed = Arrays.asList(-4, -3, -2, -1, 0, 1, 2, 3, 4);

        List<Integer> positiveEvens = mixed.stream()
                .filter(isEven.and(isPositive)) // both must be true
                .collect(Collectors.toList());

        System.out.println("Positive evens: " + positiveEvens);
    }
}
