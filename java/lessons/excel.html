Data tableTurn an Excel sheet into a DATA TABLE: a List of Maps, one Map per data row, keyed by the header names.

DataTable.java

package com.sdet.studyplan.excel;

import java.io.FileInputStream;
import java.io.IOException;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

import org.apache.poi.ss.usermodel.DataFormatter;
import org.apache.poi.ss.usermodel.Row;
import org.apache.poi.xssf.usermodel.XSSFSheet;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

/**
 * Turn an Excel sheet into a DATA TABLE: a List of Maps, one Map per
 * data row, keyed by the header names.
 *
 * For Apache.xlsx:
 *   Row 0 (header): Username | Password
 *   Row 1:          Balaji   | Sannasi#5835
 *   Row 2:          Yugan    | 123557
 *
 * becomes:
 *   [ {Username=Balaji, Password=Sannasi#5835},
 *     {Username=Yugan,  Password=123557} ]
 *
 * This is THE standard shape for data-driven testing: hand the list to
 * a TestNG @DataProvider and each Map becomes one test run — one login
 * attempt per row, with the code never hardcoding the values.
 */
public class DataTable {

    /**
     * Reads the first sheet of the given .xlsx file and returns one Map
     * per data row, where each key is the header of that column.
     */
    public static List<Map<String, String>> readAsTable(String fileName) throws IOException {
        List<Map<String, String>> table = new ArrayList<>();

        try (FileInputStream file = new FileInputStream(fileName);
                XSSFWorkbook workbook = new XSSFWorkbook(file)) {

            XSSFSheet sheet = workbook.getSheetAt(0);

            // DataFormatter gives the text as Excel displays it — without
            // it, a numeric cell like 123557 reads back as "123557.0".
            DataFormatter formatter = new DataFormatter();

            // Step 1: read the header row — it provides the map keys.
            Row headerRow = sheet.getRow(0);
            List<String> headers = new ArrayList<>();
            for (int j = 0; j < headerRow.getPhysicalNumberOfCells(); j++) {
                headers.add(formatter.formatCellValue(headerRow.getCell(j)));
            }

            // Step 2: for each data row, pair every cell with its header.
            // getLastRowNum() is the index of the final row. Looping to
            // getPhysicalNumberOfRows() instead silently drops trailing rows
            // whenever the sheet has a blank line in the middle, because the
            // physical count is smaller than the last index.
            for (int i = 1; i <= sheet.getLastRowNum(); i++) {
                Row row = sheet.getRow(i);
                if (row == null) {
                    continue;
                }

                Map<String, String> rowData = new LinkedHashMap<>();
                for (int j = 0; j < headers.size(); j++) {
                    String cellValue = row.getCell(j) == null ? "" : formatter.formatCellValue(row.getCell(j));
                    rowData.put(headers.get(j), cellValue);
                }
                table.add(rowData);
            }
        }

        return table;
    }

    public static void main(String[] args) throws IOException {
        List<Map<String, String>> table = readAsTable("Apache.xlsx");

        // Reading the data back by column name — no magic column indexes.
        for (Map<String, String> row : table) {
            System.out.println("username=" + row.get("Username")
                    + ", password=" + row.get("Password"));
        }
    }
}

package com.sdet.studyplan.excel;

import java.io.FileInputStream;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;

import org.apache.poi.ss.usermodel.Cell;
import org.apache.poi.ss.usermodel.DataFormatter;
import org.apache.poi.ss.usermodel.Row;
import org.apache.poi.xssf.usermodel.XSSFSheet;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

/**
 * Read and print every row of an .xlsx file with Apache POI.
 *
 * The POI vocabulary maps 1:1 to what you see in Excel:
 *   Workbook -> the whole file          (XSSFWorkbook = .xlsx format)
 *   Sheet    -> one tab at the bottom   (XSSFSheet)
 *   Row      -> one horizontal line     (Row)
 *   Cell     -> one box in the grid     (Cell)
 *
 * Why keep test data in Excel at all? So data can change (new logins,
 * new environments) without touching the test code.
 *
 * try-with-resources note: "try (XSSFWorkbook workbook = ...)" closes
 * the workbook and the file automatically when the block ends — no
 * manual close() needed, even if an exception is thrown.
 */
public class ReadExcel {

    public static void main(String[] args) throws IOException {
        Path excelPath = Path.of("Apache.xlsx");
        if (Files.notExists(excelPath)) {
            throw new IOException("Excel file not found: " + excelPath.toAbsolutePath());
        }

        try (FileInputStream file = new FileInputStream(excelPath.toFile());
                XSSFWorkbook workbook = new XSSFWorkbook(file)) {

            XSSFSheet sheet = workbook.getSheetAt(0); // first tab

            // DataFormatter renders a cell the way Excel DISPLAYS it.
            // We need it because numeric cells come back as doubles:
            // cell.toString() on the password 123557 would print
            // "123557.0" — a wrong password in a real login test!
            DataFormatter formatter = new DataFormatter();

            // getPhysicalNumberOfRows counts only rows that actually have
            // content — safer than getLastRowNum() for typical files.
            int rowCount = sheet.getPhysicalNumberOfRows();

            // Start at row 1: row 0 is the header (Username | Password).
            for (int i = 1; i < rowCount; i++) {
                Row row = sheet.getRow(i);
                if (row == null) {
                    continue; // blank line inside the sheet
                }

                for (int j = 0; j < row.getPhysicalNumberOfCells(); j++) {
                    Cell cell = row.getCell(j);
                    if (cell != null) {
                        System.out.print(formatter.formatCellValue(cell) + " ");
                    }
                }
                System.out.println();
            }
        }
    }
}
